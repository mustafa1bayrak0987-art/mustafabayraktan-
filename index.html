import pygame import math import random pygame.init() WIDTH, HEIGHT = 900, 900 screen = pygame.display.set_mode((WIDTH, HEIGHT)) pygame.display.set_caption("Python Heart") clock = pygame.time.Clock() CX = WIDTH // 2 CY = HEIGHT // 2 # Kalp noktalarını oluştur points = [] for i in range(500): t = (2 * math.pi * i) / 500 x = 16 * math.sin(t) ** 3 y = (13 * math.cos(t) - 5 * math.cos(2 * t) - 2 * math.cos(3 * t) - math.cos(4 * t)) scale = 24 px = CX + x * scale py = CY - y * scale points.append((px, py)) running = True angle = 0 while running: for event in pygame.event.get(): if event.type == pygame.QUIT: running = False screen.fill((3, 3, 10)) # Kalp çizgileri for i, (x, y) in enumerate(points): # Hafif dönme efekti dx = x - CX dy = y - CY a = math.radians(angle) rx = dx * math.cos(a) - dy * math.sin(a) ry = dx * math.sin(a) + dy * math.cos(a) end_x = CX + rx end_y = CY + ry # Mor -> mavi geçiş color_value = int(120 + 100 * math.sin(i * 0.04 + angle * 0.03)) color = (80, 60, max(150, color_value)) # Merkezden kalp kenarına çizgi pygame.draw.line(screen, color, (CX, CY), (end_x, end_y), 1) # Ortadaki parlak nokta pygame.draw.circle(screen, (130, 180, 255), (CX, CY), 5) pygame.display.flip() angle += 0.35 clock.tick(60) pygame.quit()
